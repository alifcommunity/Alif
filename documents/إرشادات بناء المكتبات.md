### ويندوز:

1. يتم إنشاء مجلد للمكتبة يحتوي التالي:
	1. مجلد include يمكن نسخه من مجلد لغة ألف
	2. مجلد include_other وهذا خاص بالمكتبة التي نحتاج إلى دمجها
	3. ملف alif.lib يمكن نسخه من مجلد لغة ألف
	4. ملف other.lib وهو ملف المكتبة التي نحتاج إلى دمجها او استخدامها ضمن main_module.cpp
	5. ملف المكتبة ألف main_module.cpp
	يتم تشغيل الطرفية x64 Native Tools Command Prompt VS  ضمن المجلد وكتابة الأمر التالي:
	```
	cl /std:c++20 /EHsc /O2 /MD /LD /TP /GL main_module.cpp
	/I include /I include_other
	/link /LTCG /LIBPATH:"." alif.lib other.lib /OUT:رسوميات.alifd
	```
	هنا /LIBPATH:"." يتم تحديد مسار ملفات .lib
	في حال إحتاجت المكتبات .lib إلى أي ملفات للبناء يجب تضمينها مثال: user32.lib
	البناء سينتج ملف رسوميات.alifd حيث يتم نسخه إلى المسار الذي يمكن استيرادها منه مع تضمين ملف other.dll في مكان التنفيذ مع ملف رسوميات.alifd عند وجود مكتبات مضافة
مثال مبسط للبناء بدون مكتبات إضافية:
```
	cl /std:c++20 /EHsc /O2 /MD /LD /TP /GL main_module.cpp
	/I include
	/link /LTCG /LIBPATH:"." alif.lib /OUT:حساب.alifd
```

2. يتم إنشاء مجلد للمكتبة يحتوي التالي:

	2. مجلد include_other وهذا خاص بالمكتبة التي نحتاج إلى دمجها
	3. ملف المكتبة main_module.cpp
	4. ملف المكتبة الخارجية libother.a او libother.so 
	ويتم تشغيل الطرفية ضمن المجلد وكتابة الأمر التالي:
	```
	c++ -std=c++20 -O3 -shared -fPIC 
	-I"include" -I"include_other" 
	main_module.cpp -o رسوميات.so -L. -lother -lX11
	```
	حيث -L. تحدد مسار احتواء ملفات .dylib او .a اثناء بناء البرنامج
	حيث -Wl,-rpath,. تقوم بتمرير المسار الحالي إلى رابط الملفات لتحديد مسار الملفات .a او .so
	في حال إحتاجت المكتبات .so إلى أي ملفات للبناء يجب تضمينها مثال: -lX11
	وهذا سينتج ملف رسوميات .so حيث يتم نسخه إلى المسار الذي يمكن استيراده منه
مثال مبسط للبناء بدون مكتبات إضافية:
```
c++ -std=c++20 -O3 -shared -fPIC 
-I"include" 
main_module.cpp -o رسوميات.so
```

##### مثال لملف main_module.cpp معقد يضم مكتبة الرسوميات Raven:

``` cpp
#include <alif.h>
#include <string.h>          /* strcmp */
#include <Raven.h>

static AlifInterpreter* g_main_interpreter = nullptr;
static AlifThread* t_thread_state = nullptr;


/* ---- Alif wrapper types ------------------------------------------- */

typedef struct {
    ALIFOBJECT_HEAD;
    Rid id;
    int orientation;     /* R_WIDGET_VBOX or R_WIDGET_HBOX */
} LayerObject;

typedef struct {
    ALIFOBJECT_HEAD;
    Rid id;
    AlifObject* callback;  /* owned Alif callable, or nullptr */
} ButtonObject;

typedef struct {
    ALIFOBJECT_HEAD;
    Rid id;
    Rid display;
} LabelObject;

/* forward declaration so Layer_add can reference it */
extern AlifTypeObject LayerType;
extern AlifTypeObject LabelType;

/* ---- trampoline: Raven (Cpp thread) -> Alif -------------------------- */

static void button_clicked_cb(Rid button, void* user) {
    ButtonObject* self = (ButtonObject*)user;
    if (!self->callback) return;

    if (t_thread_state == nullptr) {
        t_thread_state = alifThreadState_new(g_main_interpreter);
    }

    alifThreadState_swap(t_thread_state);

    alifObject_callFunction(self->callback, "O", (AlifObject*)self);
    if (alifErr_occurred()) alifErr_print();

    alifThreadState_swap(nullptr);
}

/* ---- Button type ------------------------------------------------------ */

static AlifObject* Button_new(AlifTypeObject* type, AlifObject* args, AlifObject* kw) {
    return type->alloc(type, 0);
}

static int Button_init(ButtonObject* self, AlifObject* args, AlifObject* kw) {
    const char* text = "زر";
    static char* kwlist[] = { (char*)"نص", nullptr };
    if (!alifArg_parseTupleAndKeywords(args, kw, "|s", kwlist, &text))
        return -1;

    self->id = Rbutton_create(text);
    self->callback = nullptr;

    Rbutton_set_clicked_callback(self->id, button_clicked_cb, self);
    ALIF_INCREF(self);
    return 0;
}

static AlifObject* Button_set_text(ButtonObject* self, AlifObject* arg) {
    const char* text = alifUStr_asUTF8(arg);
    if (!text) return nullptr;
    Rbutton_set_text(self->id, text);
    ALIF_RETURN_NONE;
}

static AlifObject* Button_set_size(ButtonObject* self, AlifObject* args) {
    float width{}, height{};
    if (!alifArg_parseTuple(args, "ff", &width, &height))
        return nullptr;
    Rwidget_set_size(self->id, width, height);
    ALIF_RETURN_NONE;
}

static AlifObject* Button_set_style(ButtonObject* self, AlifObject* arg) {
    const char* s = alifUStr_asUTF8(arg);
    if (!s) return nullptr;

    if (strcmp(s, "اساسي") == 0)
        Rbutton_set_style(self->id, R_BUTTON_PRIMARY);
    else if (strcmp(s, "مستوي") == 0)
        Rbutton_set_style(self->id, R_BUTTON_FLAT);
    else
        Rbutton_set_style(self->id, R_BUTTON_DEFAULT);
    ALIF_RETURN_NONE;
}

static AlifObject* Button_on_clicked(ButtonObject* self, AlifObject* arg) {
    if (!alifCallable_check(arg)) {
        alifErr_setString(_alifExcTypeError_, "يتوقع وسيط قابل للإستدعاء");
        return nullptr;
    }
    ALIF_XINCREF(arg);
    ALIF_XDECREF(self->callback);
    self->callback = arg;
    ALIF_RETURN_NONE;
}

static void Button_dealloc(ButtonObject* self) {
    ALIF_XDECREF(self->callback);
    ALIF_TYPE(self)->free((AlifObject*)self);
}

static AlifMethodDef _buttonMethods_[] = {
    {"اضبط_نص",   (AlifCPPFunction)Button_set_text,   METHOD_O},
    {"اضبط_الحجم", (AlifCPPFunction)Button_set_size, METHOD_VARARGS},
    {"اضبط_المظهر", (AlifCPPFunction)Button_set_style, METHOD_O},
    {"عند_الضغط", (AlifCPPFunction)Button_on_clicked, METHOD_O},
    {nullptr, nullptr}
};

AlifTypeObject ButtonType = {
    .objBase = ALIFVAROBJECT_HEAD_INIT(nullptr, 0),
    .name = "رسوميات.زر",
    .basicSize = sizeof(ButtonObject),
    .dealloc = (Destructor)Button_dealloc,
    .flags = ALIF_TPFLAGS_DEFAULT,
    .methods = _buttonMethods_,
    .init = (InitProc)Button_init,
    .new_ = Button_new,
};

/* ---- Layer type (VBox / HBox) ----------------------------------------- */

static AlifObject* Layer_new(AlifTypeObject* type, AlifObject* args, AlifObject* kw) {
    return type->alloc(type, 0);
}

static int Layer_init(LayerObject* self, AlifObject* args, AlifObject* kw) {
    const char* orientation = "عمودي";
    float spacing = 8.0f;
    static char* kwlist[] = { (char*)"دوران", (char*)"مسافة", nullptr };
    if (!alifArg_parseTupleAndKeywords(args, kw, "|sf", kwlist,
        &orientation, &spacing))
        return -1;

    self->id = Rwidget_create(
        strcmp(orientation, "افقي") == 0
        ? R_WIDGET_HBOX : R_WIDGET_VBOX);
    Rpanel_set_spacing(self->id, spacing);
    Rslot_set_halign(self->id, R_HALIGN_CENTER);
    Rslot_set_valign(self->id, R_VALIGN_CENTER);
    return 0;
}

static AlifObject* Layer_add(LayerObject* self, AlifObject* arg) {
    if (alifObject_typeCheck(arg, &ButtonType)) {
        Rwidget_add_child(self->id, ((ButtonObject*)arg)->id);
    }
    else if (alifObject_typeCheck(arg, &LayerType)) {
        Rwidget_add_child(self->id, ((LayerObject*)arg)->id);
    }
    else if (alifObject_typeCheck(arg, &LabelType)) {
        Rwidget_add_child(self->id, ((LabelObject*)arg)->id);
    }
    else {
        alifErr_setString(_alifExcTypeError_, "طبقة.اضف() تحتاج زر او طبقة او رقعة");
        return nullptr;
    }
    ALIF_RETURN_NONE;
}

static AlifMethodDef _layerMethods_[] = {
    {"اضف", (AlifCPPFunction)Layer_add, METHOD_O},
    {nullptr, nullptr}
};

AlifTypeObject LayerType = {

static AlifObject* alif_init(AlifObject* self, AlifObject* ALIF_UNUSED(ignored)) {
    Rinit(R_BACKEND_DEFAULT);
    Rtheme_apply(R_THEME_GLASS_LIGHT);
    ALIF_RETURN_NONE;
}

static AlifObject* alif_window(AlifObject* self, AlifObject* args, AlifObject* kw) {
    const char* title = "ألف";
    int w = 640, h = 480;
    static char* kwlist[] = { (char*)"title", (char*)"width", (char*)"height", nullptr };
    if (!alifArg_parseTupleAndKeywords(args, kw, "|sii", kwlist,
        &title, &w, &h))
        return nullptr;

    Rid win = Rwindow_create(title, w, h, R_WINDOW_DEFAULT);
    Rwindow_set_background(win, Rtexture_create_from_file("image.jpg"));
    return alifLong_fromUnsignedLongLong((unsigned long long)win);
}

static AlifObject* alif_set_content(AlifObject* self, AlifObject* args) {
    unsigned long long win;
    AlifObject* layer;
    if (!alifArg_parseTuple(args, "KO", &win, &layer)) return nullptr;
    if (!alifObject_typeCheck(layer, &LayerType)) {
        alifErr_setString(_alifExcTypeError_, "المحتوى يجب أن يكون طبقة");
        return nullptr;
    }
    Rwindow_set_content((Rid)win, ((LayerObject*)layer)->id);
    ALIF_RETURN_NONE;
}

static AlifObject* alif_run(AlifObject* self, AlifObject* ALIF_UNUSED(ignored)) {
    ALIF_BEGIN_ALLOW_THREADS
        while (Rwait_frame(-1.0)) {}
    Rshutdown();
    ALIF_END_ALLOW_THREADS
    ALIF_RETURN_NONE;
}

static AlifObject* alif_quit(AlifObject* self, AlifObject* ALIF_UNUSED(ignored)) {
    Rquit(0);
    ALIF_RETURN_NONE;
}

/* ---- multi-phase init: exec slot adds the types ----------------------- */

static int ravengui_exec(AlifObject* m) {
    // Save main interpreter state reference
    g_main_interpreter = alifInterpreterState_main();

    if (alifType_ready(&LayerType) < 0)
        return -1;
    if (alifType_ready(&ButtonType) < 0)
        return -1;
    if (alifType_ready(&LabelType) < 0)
        return -1;

    if (alifModule_addObjectRef(m, "طبقة", (AlifObject*)&LayerType) < 0)
        return -1;
    if (alifModule_addObjectRef(m, "زر", (AlifObject*)&ButtonType) < 0)
        return -1;
    if (alifModule_addObjectRef(m, "رقعة", (AlifObject*)&LabelType) < 0)
        return -1;
    return 0;
}

static AlifModuleDefSlot _ravenGUISlots_[] = {
    {ALIF_MOD_EXEC, (void*)ravengui_exec},
    {0, nullptr},
};

static AlifModuleDef _ravenGUIModule_ = {
    ALIFMODULEDEF_HEAD_INIT,
    "رسوميات",
    "مكتبة رسوميات Raven مع ألف",
    0,
    _moduleMethods_,
    _ravenGUISlots_,
};

ALIFMODINIT_FUNC alifInitU_mgbgti4grap(void) {
    /* multi-phase: return the module def; Alif calls the exec slot */
    return alifModuleDef_init(&_ravenGUIModule_);
}
```
